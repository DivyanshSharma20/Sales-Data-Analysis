--TO SHOW WHICH CATEGORY HAVE EARNED MOST--

SELECT CATEGORY,SUM(SALES) AS total_sales
FROM SALES
GROUP BY CATEGORY;

--TO SHOW WHICH CATEGORY HAVE MAX SALE BY SUB-CATEGORY WISE--

SELECT CATEGORY ,MAX(SALES),`Sub-Category`
FROM SALES
WHERE CATEGORY="FURNITURE" 
GROUP BY `SUB-CATEGORY`
ORDER BY MAX(SALES) DESC;

SELECT CATEGORY ,MAX(SALES),`Sub-Category`
FROM SALES
WHERE CATEGORY="TECHNOLOGY" 
GROUP BY `SUB-CATEGORY`
ORDER BY MAX(SALES) DESC;

SELECT CATEGORY ,MAX(SALES),`Sub-Category`
FROM SALES
WHERE CATEGORY="OFFICE SUPPLIES" 
GROUP BY `SUB-CATEGORY`
ORDER BY MAX(SALES) DESC;

--TO SHOW TOP 5 MOST PROFITABLE PRODUCTS AND THERE COUNT--

SELECT `PRODUCT NAME`,SUM(PROFIT) AS TOTAL_PROFITS,COUNT(`PRODUCT NAME`)
FROM SALES
GROUP BY `PRODUCT NAME`
ORDER BY SUM(PROFIT) DESC; 

--TO FIND WHICH DATE HAS GOT MOST ORDERS--

SELECT `ORDER DATE`,COUNT(`ORDER ID`)AS TOTAL_ORDERS
FROM SALES
GROUP BY `ORDER DATE`
ORDER BY COUNT(`ORDER ID`) DESC;

--TO FIND TOTAL SALES PER CITY--

SELECT SUM(SALES) AS TOTAL_SALES_EACH_CITY ,CITY
FROM SALES
GROUP BY CITY 
ORDER BY  TOTAL_SALES_EACH_CITY DESC;

--PROFIT BY CATEGORY--
SELECT Category, SUM(Profit) AS Total_Profit
FROM sales
GROUP BY Category
ORDER BY Total_Profit DESC;

--HOW MANY ORDERS BY SHIP MODE--
SELECT `SHIP MODE`,COUNT(`ROW ID`) AS TOTAL_ORDERS
FROM SALES
GROUP BY `SHIP MODE`
ORDER BY TOTAL_ORDERS DESC;

--TOP !) PRODUCT BY SALES--
SELECT `PRODUCT NAME`,SUM(SALES)AS TOTAL_SALES
FROM SALES 
GROUP BY `PRODUCT NAME`
ORDER BY TOTAL_SALES asc
LIMIT 10; 

--LOSS MAKING PRODUCTS--
SELECT `PRODUCT NAME`,SUM(PROFIT)AS TOTAL_PROFIT
FROM SALES
GROUP BY `PRODUCT NAME`
HAVING TOTAL_PROFIT <0
ORDER BY TOTAL_PROFIT;

--TOP 10 MOST PROFITABLE CUSTOMER--
SELECT `CUSTOMER NAME` , SUM(PROFIT)AS TOTAL_PROFIT
FROM SALES
GROUP BY `CUSTOMER NAME`
ORDER BY SUM(PROFIT) DESC
LIMIT 10;

--Customers Causing Loss--
SELECT `CUSTOMER NAME`,
            SUM(PROFIT)AS TOTAL_PROFIT
            FROM SALES
            GROUP BY `CUSTOMER NAME`
            HAVING TOTAL_PROFIT<0
            LIMIT 5;

--MARGIN PROFIT--
SELECT CATEGORY,SUM(PROFIT)/SUM(SALES)*100 AS MARGIN_PROFIT
FROM SALES
GROUP BY CATEGORY
ORDER BY MARGIN_PROFIT ASC;

--Yearly Sales Trend--
SELECT YEAR(`Order Date`) AS Year,
       SUM(Sales) AS Yearly_Sales
FROM sales
GROUP BY Year;


--Rank Products by Sales--
SELECT `Product Name`,
       SUM(Sales) AS Total_Sales,
       RANK() OVER(ORDER BY SUM(Sales) DESC) AS Ranking
FROM sales
GROUP BY `Product Name`;

--WHICH PRODUCT SHOULD BE REMOVED--
SELECT `Product Name`,
       SUM(Profit) AS Total_Profit
FROM sales
GROUP BY `Product Name`
HAVING Total_Profit < 0;


--WHICH REGION SHOULD FOCUSED ON BASIS OF PROFIT--
SELECT REGION , SUM(PROFIT)AS TOTAL_PROFIT
FROM SALES
GROUP BY REGION
ORDER BY TOTAL_PROFIT DESC;

--DOES DISCOUNT AFFECT PROFIT--
SELECT DISCOUNT,AVG(PROFIT)AS AVG_PROFIT
FROM SALES
GROUP BY DISCOUNT
ORDER BY DISCOUNT ASC;
